```typescript
import { supabase } from './supabase';
import type { Session } from '../types';

export async function fetchActiveSessions(groupId: string): Promise<Session[]> {
  const { data, error } = await supabase
    .from('sessions')
    .select('*')
    .eq('group_id', groupId)
    .eq('status', 'active')
    .order('started_at', { ascending: false });

  if (error) throw new Error(error.message);
  return data || [];
}

export async function fetchSessionDetails(sessionId: string): Promise<Session> {
  const { data, error } = await supabase
    .from('sessions')
    .select('*, rounds(*, round_items(*, consumptions(*))), payments(*), photos(*)')
    .eq('id', sessionId)
    .single();

  if (error) throw new Error(error.message);
  return data;
}
```

```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { fetchActiveSessions, createSession } from '../api/sessions';

export function useSession(groupId: string) {
  return useQuery({
    queryKey: ['sessions', groupId],
    queryFn: () => fetchActiveSessions(groupId),
    staleTime: 30 * 1000, // 30s
  });
}

export function useCreateSession() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: createSession,
    onSuccess: (newSession) => {
      queryClient.setQueryData(['sessions', newSession.group_id], (old) => [
        newSession,
        ...(old || []),
      ]);
    },
  });
}
```

```typescript
import { View, ScrollView, Text, Pressable, FlatList } from 'react-native';
import { useActiveSession } from '../src/hooks/useSession';
import { useRounds } from '../src/hooks/useRounds';
import { usePayments } from '../src/hooks/usePayments';
import { RoundItem } from '../src/components/mobile/RoundItem';
import { MemberDebt } from '../src/components/mobile/MemberDebt';
import { computeSessionTotals } from '../src/domain/debt';

export default function NiteScreen() {
  const { data: session } = useActiveSession();
  const { data: rounds } = useRounds(session?.id);

  if (!session) return <Text className="text-white">Nenhuma noite ativa</Text>;

  const totals = computeSessionTotals(session);

  return (
    <ScrollView className="flex-1 bg-ink px-4 pt-4">
      <Text className="text-white text-2xl font-black mb-4">{session.name}</Text>

      <Text className="text-white/60 text-sm mb-2">Rondas</Text>
      <FlatList
        data={rounds}
        keyExtractor={(r) => r.id}
        renderItem={({ item }) => <RoundItem round={item} />}
        scrollEnabled={false}
      />

      <Pressable className="bg-brand rounded-2xl px-4 py-4 mt-4">
        <Text className="text-black font-black text-center">Nova ronda</Text>
      </Pressable>

      <Text className="text-white/60 text-sm mb-2 mt-6">Dívidas</Text>
      {totals.perMember.map((debt) => (
        <MemberDebt key={debt.memberId} debt={debt} />
      ))}
    </ScrollView>
  );
}
```

```typescript
export function useRealtimeRounds(sessionId: string) {
  const queryClient = useQueryClient();

  useEffect(() => {
    const channel = supabase
      .channel(`rounds:${sessionId}`)
      .on(
        'postgres_changes',
        { event: 'INSERT', schema: 'public', table: 'rounds', filter: `session_id=eq.${sessionId}` },
        (payload) => {
          queryClient.invalidateQueries({ queryKey: ['rounds', sessionId] });
        }
      )
      .subscribe();

    return () => supabase.removeChannel(channel);
  }, [sessionId, queryClient]);
}
```
